cmake_minimum_required(VERSION 3.28)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(obs-midikeyboard-tests LANGUAGES CXX)
  enable_testing()
endif()

add_executable(midi-chord-tests chord-tests.cpp ../src/midi-state.cpp ../src/midi-device.cpp ../src/chord-detector.cpp)
target_include_directories(midi-chord-tests PRIVATE ../src)
target_compile_features(midi-chord-tests PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(midi-chord-tests PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(midi-chord-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME midi-chord-tests COMMAND midi-chord-tests)

add_executable(piano-layout-tests piano-layout-tests.cpp ../src/piano-layout.cpp)
target_include_directories(piano-layout-tests PRIVATE ../src)
target_compile_features(piano-layout-tests PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(piano-layout-tests PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(piano-layout-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME piano-layout-tests COMMAND piano-layout-tests)

option(BUILD_OBS_SMOKE_TESTS "Build the optional integration test for an installed OBS runtime" OFF)
if(BUILD_OBS_SMOKE_TESTS)
  if(NOT WIN32)
    message(FATAL_ERROR "The optional OBS runtime smoke test currently supports Windows only")
  endif()
  if(NOT TARGET OBS::libobs)
    find_package(libobs REQUIRED)
  endif()
  add_executable(obs-chord-smoke obs-smoke.cpp)
  target_compile_features(obs-chord-smoke PRIVATE cxx_std_20)
  target_link_libraries(obs-chord-smoke PRIVATE OBS::libobs)
  if(MSVC)
    target_compile_options(obs-chord-smoke PRIVATE /W4 /WX /utf-8)
  else()
    target_compile_options(obs-chord-smoke PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endif()
